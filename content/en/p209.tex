\VerdierSemanticAnchor{ch3d-3-1-3-7}
\[
\textit{(7) For every $X$ in $K^{b}(\mathcal A)$ and every $Y$ in
$K^{+}(\mathcal A)$,}\qquad
\operatorname{Ext}_{\mathcal A}^{0}(X,Y)\simeq
\varinjlim_{(\operatorname{Qis}^{b}/X)^o}
\operatorname{Hom}_{K(\mathcal A)}(\mathord\cdot,Y).
\]

\VerdierSemanticAnchor{ch3d-3-1-3-8}
\[
\textit{(8) For every $X$ in $K^{-}(\mathcal A)$ and every $Y$ in
$K^{b}(\mathcal A)$,}\qquad
\operatorname{Ext}_{\mathcal A}^{0}(X,Y)\simeq
\varinjlim_{Y\backslash\operatorname{Qis}^{b}}
\operatorname{Hom}_{K(\mathcal A)}(X,\mathord\cdot).
\]

\VerdierSemanticAnchor{ch3d-3-1-3-proof}
Formula (1) follows from the definition of the localized category (Chap. II,
2.2.2). Formula (2) follows from formula (1) by passing to the opposite
category (Chap. II, 2.2.4). Formula (3) then follows from formulas (1) and
(2). Formula (4) is a definition (Chap. I, 1.3.8). Formulas (5) and (6)
follow from Proposition (1.2.1), b) and b)\textsuperscript{bis}. Formulas
(7) and (8) follow from Proposition (1.2.1), c) and
c)\textsuperscript{bis}.

\VerdierSemanticAnchor{ch3d-3-1-4}
\medskip\noindent\textbf{Proposition 3.1.4.}\quad
\textit{Suppose that every object of the category $\mathcal A$ is isomorphic
to a subobject (respectively, to a quotient) of an injective (respectively,
projective) object of $\mathcal A$.}

\VerdierSemanticAnchor{ch3d-3-1-4-1}
\noindent\textit{1) Every object $Y$ (respectively, $X$) of
$K^{+}(\mathcal A)$ (respectively, $K^{-}(\mathcal A)$) has a right
(respectively, left) resolution $u\colon Y\longrightarrow I^\bullet$
(respectively, $u\colon P^\bullet\longrightarrow X$) of injective
(respectively, projective) type (1.1.4).}

\VerdierSemanticAnchor{ch3d-3-1-4-2}
\noindent\textit{2) This resolution defines an isomorphism:}
\[
\operatorname{Ext}_{\mathcal A}^{0}(X,Y)\simeq
\operatorname{Hom}_{K(\mathcal A)}(X,I^\bullet),
\qquad\textit{for every $X$ in $K(\mathcal A)$}
\]
\noindent\textit{(respectively,}
\[
\operatorname{Ext}_{\mathcal A}^{0}(X,Y)\simeq
\operatorname{Hom}_{K(\mathcal A)}(P^\bullet,Y),
\qquad\textit{for every $Y$ in $K(\mathcal A)$).}
\]

\VerdierSemanticAnchor{ch3d-3-1-4-3}
\noindent\textit{3) The cohomological dimension of $\mathcal A$ is the
supremum of the injective (respectively, projective) dimensions of the
objects of $\mathcal A$.}

\VerdierSemanticAnchor{ch3d-3-1-4-4}
\noindent\textit{4) When the cohomological dimension of $\mathcal A$ is
finite, every object $Y$ (respectively, $X$) of $K(\mathcal A)$ has a right
(respectively, left) resolution $u\colon Y\longrightarrow I^\bullet$
(respectively, $u\colon P^\bullet\longrightarrow X$) of injective
(respectively, projective) type.}

\VerdierSemanticAnchor{ch3d-3-1-4-5}
\noindent\textit{5) Under the hypotheses and notation of 4), such a
resolution defines an isomorphism:}
\[
\operatorname{Ext}_{\mathcal A}^{0}(X,Y)\simeq
\operatorname{Hom}_{K(\mathcal A)}(X,I^\bullet),
\qquad\textit{for every $X$ in $K(\mathcal A)$}
\]
\noindent\textit{(respectively,}
\[
\operatorname{Ext}_{\mathcal A}^{0}(X,Y)\simeq
\operatorname{Hom}_{K(\mathcal A)}(P^\bullet,Y),
\qquad\textit{for every $Y$ in $K(\mathcal A)$).}
\]
