Finally, for every element $(p,q)$ of
$\mathcal{F}\ell_1(\widetilde{\mathbb Z})$, denote by
\begin{equation}
J(p,q)\colon\mathcal Y(p,q)\longrightarrow Y(p,q)
\tag{4.1.2.9}
\end{equation}
the homotopy class of the morphism of complexes

\VerdierSemanticAnchor{ch4d-4-1-2-10}
\begin{equation}
j(p,q)\colon\mathcal Y(p,q)\longrightarrow Y(p,q),
\tag{4.1.2.10}
\end{equation}
defined by the formulas

\VerdierSemanticAnchor{ch4d-4-1-2-11}
\begin{equation}
\begin{aligned}
j(p,q)^i&=0,
  &&i<-q\ \lor\ i>-p-1,\\
j(p,q)^i&=\operatorname{id},
  &&-q\leq i\leq-p-2,\\
j(p,q)^{-p-1}&\colon
  Y^{-p-1}\oplus Y^{-p}\longrightarrow Y^{-p-1},\\
j(p,q)^{-p-1}&=(\operatorname{id},0).
\end{aligned}
\tag{4.1.2.11}
\end{equation}

\VerdierSemanticAnchor{ch4d-4-1-3}
\medskip\noindent\textbf{Proposition 4.1.3.}\quad
\textit{a) For every element $(p,q)$ of
$\mathcal{F}\ell_1(\widetilde{\mathbb Z})$, the morphism
$J(p,q)\colon\mathcal Y(p,q)\longrightarrow Y(p,q)$ of $K(\mathcal A)$ is
an isomorphism.}

\VerdierSemanticAnchor{ch4d-4-1-3-b}
\noindent\textit{b) For every pair $(p,q)$, $(p',q')$ of
$\mathcal{F}\ell_1(\widetilde{\mathbb Z})$ such that $p\leq p'$ and
$q\leq q'$, the following diagram of $K(\mathcal A)$:}

\VerdierSemanticAnchor{ch4d-4-1-3-b-diagram}
\begin{center}
\begin{tikzcd}[column sep=7.0em,row sep=3.2em]
\mathcal Y(p,q)
  \arrow[r,"{\Phi((p,q),(p',q'))}"]
  \arrow[d,"{J(p,q)}"']
&\mathcal Y(p',q') \arrow[d,"{J(p',q')}"]\\
Y(p,q)
  \arrow[r,"{\varphi((p,q),(p',q'))}"']
&Y(p',q')
\end{tikzcd}
\end{center}
\noindent\textit{is commutative.}

\VerdierSemanticAnchor{ch4d-4-1-3-c}
\noindent\textit{c) For every element $(p,q,r)$ of
$\mathcal{F}\ell_2(\widetilde{\mathbb Z})$, the following diagram of
$K(\mathcal A)$:}

\VerdierSemanticAnchor{ch4d-4-1-3-c-diagram}
\begin{center}
\begin{tikzcd}[column sep=7.0em,row sep=3.2em]
\mathcal Y(q,r)
  \arrow[r,"{\delta(p,q,r)}"]
  \arrow[d,"{J(q,r)}"']
&\mathcal Y(p,q)[1] \arrow[d,"{J(p,q)[1]}"]\\
Y(q,r)
  \arrow[r,"{\delta(p,q,r)}"']
&Y(p,q)[1]
\end{tikzcd}
\end{center}
\noindent\textit{is commutative.}
