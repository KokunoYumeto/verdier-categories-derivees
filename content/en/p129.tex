\VerdierSemanticAnchor{ch2-2-1-1-sm3-conditions}
\VerdierSemanticAnchor{ch2-2-1-1-sm3-i}
\noindent\textit{i)} There exists a morphism $s$ in $S$, with source $Y$,
such that $sf=sg$.

\VerdierSemanticAnchor{ch2-2-1-1-sm3-ii}
\noindent\textit{ii)} There exists a morphism $t$ in $S$, with target $X$,
such that $ft=gt$.

\VerdierSemanticAnchor{ch2-2-1-1-saturated}
\noindent A multiplicative system is said to be \emph{saturated} if it has the
following property:

\VerdierSemanticAnchor{ch2-2-1-1-sm4}
\VerdierSemanticAnchor{ch2-2-1-1-saturated-sm4}
\smallskip
\noindent\textbf{SM4)} If $f,g,$ and $h$ are three composable morphisms and if
the morphisms $fg$ and $gh$ belong to $S$, then the morphism $g$ belongs to
$S$.

\VerdierSemanticAnchor{ch2-2-1-2}
\medskip
\noindent\textbf{2.1.2.}\quad
\VerdierSemanticAnchor{ch2-2-1-2-compatible-triangulation}
\VerdierSemanticAnchor{ch2-2-1-2-compatible}
Let $\mathcal D$ be a triangulated category. A multiplicative system $S$ of
$\mathcal D$ is said to be \emph{compatible with the triangulation} if it has
the following properties:

\VerdierSemanticAnchor{ch2-2-1-2-sm5}
\smallskip
\noindent\textbf{SM5)} For every element $s$ of $S$ and every integer
$n\in\mathbb Z$, the morphism $s[n]$ belongs to $S$. In other words, $S$ is
\emph{stable under the translation automorphisms}.

\VerdierSemanticAnchor{ch2-2-1-2-sm6}
\smallskip
\noindent\textbf{SM6)} For every pair of distinguished triangles:

\VerdierSemanticAnchor{ch2-2-1-2-sm6-triangle-pair}
\[
  (X,Y,Z,u,v,w), \qquad (X',Y',Z',u',v',w')
\]
\noindent and every commutative diagram:

\VerdierSemanticAnchor{ch2-2-1-2-sm6-square}
\begin{center}
  \begin{tikzcd}[column sep=4.2em,row sep=2.5em]
    X \arrow[r,"u"] \arrow[d,"s"'] & Y \arrow[d,"s'"] \\
    X' \arrow[r,"u'"'] & Y'
  \end{tikzcd}
  \quad ,
\end{center}
\VerdierSemanticAnchor{ch2-2-1-2-sm6-triangle-morphism}
\VerdierSemanticAnchor{ch2-2-1-2-sm6-completion}
\noindent where $s$ and $s'$ are elements of $S$, there exists a morphism
belonging to $S$, $s''\colon Z\to Z'$, such that $(s,s',s'')$ is a morphism of
triangles.

\VerdierSemanticAnchor{ch2-2-1-3}
\medskip
\noindent\textbf{Remark 2.1.3.}\quad
\VerdierSemanticAnchor{ch2-2-1-3-item1}
\VerdierSemanticAnchor{ch2-2-1-3-1}
\textbf{1)} Let $\mathcal D$ be a triangulated category, let $S$ be a
multiplicative system of $\mathcal D$ compatible with the triangulation, and
let $\mathcal D^\circ$ be the triangulated category opposite to $\mathcal D$.
The set $S$ of morphisms of $\mathcal D^\circ$ is a multiplicative system of
$\mathcal D^\circ$ compatible with the triangulation of $\mathcal D^\circ$
(1.1.7).

\VerdierSemanticAnchor{ch2-2-1-3-item2}
\VerdierSemanticAnchor{ch2-2-1-3-2}
\smallskip
\noindent\textbf{2)} Let $\mathcal D$ be a triangulated category and let $S$
be a set of morphisms of $\mathcal D$ having properties SM1), SM5), and SM6).
Then the set $S$ also has property SM2).
\VerdierSemanticAnchor{ch2-2-1-3-2-sm2}
Indeed, it suffices to show that every diagram:

\VerdierSemanticAnchor{ch2-2-1-3-item2-input-boundary}
\VerdierSemanticAnchor{ch2-2-1-3-2-input-boundary}
\begin{center}
  \begin{tikzcd}[column sep=4.0em,row sep=2.6em]
    & X' \arrow[d,"s\in S"] \\
    Z \arrow[r,"f"] & X
  \end{tikzcd}
\end{center}
