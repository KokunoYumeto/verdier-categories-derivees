\VerdierSemanticAnchor{ch3d-2-4-4-b}
\noindent\textit{b) Let $P$ denote the set of projective objects of
$\mathcal A$. The triangulated category $K^{-}(P)$ is a subcategory of the
category of objects that are $Q$-free on the left. The natural functor}
\[
K^{-}(P)\longrightarrow D(\mathcal A)
\]
\noindent\textit{is fully faithful.}

\VerdierSemanticAnchor{ch3d-2-4-4-c}
\noindent\textit{c) If every object of $\mathcal A$ is a quotient of a
projective object (respectively, and if in addition there is an integer
$n\in\mathbb N$ bounding the projective dimension of the objects of
$\mathcal A$), then:}

\noindent\textit{1) The natural functor
$K^{-}(P)\longrightarrow D^{-}(\mathcal A)$ (respectively,
$K^{*}(P)\longrightarrow D^{*}(\mathcal A)$) is an equivalence of
categories. Let
$L^{-}\colon D^{-}(\mathcal A)\longrightarrow K^{-}(P)$ (respectively,
$L^{*}\colon D^{*}(\mathcal A)\longrightarrow K^{*}(P)$) denote a
quasi-inverse functor.}

\noindent\textit{2) The functor
$L^{-}Q^{-}\colon K^{-}(\mathcal A)\longrightarrow K^{-}(P)$
(respectively,
$L^{*}Q^{*}\colon K^{*}(\mathcal A)\longrightarrow K^{*}(P)$) is right
adjoint to the inclusion functor}
\[
K^{-}(P)\longrightarrow K^{-}(\mathcal A)
\qquad
(\text{respectively, }K^{*}(P)\longrightarrow K^{*}(\mathcal A)).
\]
\noindent\textit{The adjunction morphism
$\operatorname{id}_{K^{-}(\mathcal A)}\longrightarrow L^{-}Q^{-}$
(respectively,
$\operatorname{id}_{K^{*}(\mathcal A)}\longrightarrow L^{*}Q^{*}$) is a
``quasi-isomorphism''.}

\noindent\textit{3) The functor $L^{-}$ (respectively, $L^{*}$) is left
adjoint to the functor $Q^{-}$ (respectively, $Q^{*}$).}

\noindent\textit{4) An object of $K^{-}(\mathcal A)$ (respectively,
$K^{*}(\mathcal A)$) is $Q$-free on the left if and only if it is
isomorphic to an object of $K^{-}(P)$ (respectively, $K^{*}(P)$).}

\VerdierSemanticAnchor{ch3d-2-4-4-proof-a}
Let us prove a). For every object $Y$ of $\mathcal A$, we have
\[
\operatorname{Hom}_{K(\mathcal A)}(X,Y[n])=0,
\qquad\text{for }n\neq0,
\]
and consequently, since $X$ is $Q$-free on the left (Chap. II, 2.3.3),
we have
\[
\operatorname{Hom}_{D(\mathcal A)}(X,Y[n])=0,
\qquad\text{for }n\neq0.
\]
Moreover,
$\operatorname{Hom}_{D(\mathcal A)}(X,Y)
=\operatorname{Hom}_{\mathcal A}(X,Y)$ (1.2.9). Let

\VerdierSemanticAnchor{ch3d-2-4-4-proof-a-short-exact}
\[
S=0\longrightarrow Y'\xrightarrow{u}Y\xrightarrow{v}Y''
\longrightarrow0
\]
be an exact sequence in $\mathcal A$. The canonical cohomological functor
$\operatorname{comp}(\mathcal A)\longrightarrow D(\mathcal A)$ (1.3.2)
gives a distinguished triangle $(Y',Y,Y'',[u],[v],\delta(S))$. Hence there
is an exact sequence (Chap. II, 1.2.1)

\VerdierSemanticAnchor{ch3d-2-4-4-proof-a-hom-exact}
\[
\cdots\longrightarrow0\longrightarrow
\operatorname{Hom}_{D(\mathcal A)}(X,Y')\longrightarrow
\operatorname{Hom}_{D(\mathcal A)}(X,Y)\longrightarrow
\operatorname{Hom}_{D(\mathcal A)}(X,Y'')\longrightarrow0
\longrightarrow\cdots,
\]
i.e.\ an exact sequence

\VerdierSemanticAnchor{ch3d-2-4-4-proof-a-hom-a-exact}
\[
0\longrightarrow\operatorname{Hom}_{\mathcal A}(X,Y')
\longrightarrow\operatorname{Hom}_{\mathcal A}(X,Y)
\longrightarrow\operatorname{Hom}_{\mathcal A}(X,Y'')
\longrightarrow0,
\]
