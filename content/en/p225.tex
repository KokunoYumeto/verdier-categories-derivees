We have thus defined a functor
$\mathcal{F}\ell_1(\widetilde{\mathbb Z})\longrightarrow K(\mathcal A)$.

Let $(p,q,r)$ be an element of
$\mathcal{F}\ell_2(\widetilde{\mathbb Z})$. Define

\VerdierSemanticAnchor{ch4d-4-1-1-6}
\begin{equation}
\delta(p,q,r)\colon\mathcal Y(q,r)\longrightarrow\mathcal Y(p,q)[1],
\tag{4.1.1.6}
\end{equation}
to be the homotopy class of the morphism of complexes

\VerdierSemanticAnchor{ch4d-4-1-1-7}
\begin{equation}
d(p,q,r)\colon\mathcal Y(q,r)\longrightarrow\mathcal Y(p,q)[1],
\tag{4.1.1.7}
\end{equation}
defined by the formulas

\VerdierSemanticAnchor{ch4d-4-1-1-8}
\begin{equation}
d(p,q,r)^i=
\begin{pmatrix}
0 & \operatorname{id}_{Y(q)}^{i+1}\\
0 & 0
\end{pmatrix}.
\tag{4.1.1.8}
\end{equation}

We know (Chapter I, 3.3.6) that the following triangle of $K(\mathcal A)$:

\VerdierSemanticAnchor{ch4d-4-1-1-distinguished-triangle}
\begin{center}
\resizebox{\textwidth}{!}{%
$\bigl(\mathcal Y(p,q),\mathcal Y(p,r),\mathcal Y(q,r),
\Phi((p,q),(p,r)),\Phi((p,r),(q,r)),\delta(p,q,r)\bigr)$% 
}
\end{center}
is distinguished and depends functorially on $(p,q,r)$. Thus we have
constructed a spectral object $(\mathcal Y(p,q),\delta(p,q,r))$ with values
in $K(\mathcal A)$.

Let $Y\colon\widetilde{\mathbb Z}\longrightarrow
\operatorname{comp}(\mathcal A)$ and
$Y'\colon\widetilde{\mathbb Z}\longrightarrow
\operatorname{comp}(\mathcal A)$ be two functors, and let
$m\colon Y\longrightarrow Y'$ be a morphism of functors. The morphism $m$
induces a morphism of the associated spectral objects:

\VerdierSemanticAnchor{ch4d-4-1-1-9}
\begin{equation}
\mathcal M(p,q)\colon\mathcal Y(p,q)\longrightarrow\mathcal Y'(p,q).
\tag{4.1.1.9}
\end{equation}

\VerdierSemanticAnchor{ch4d-4-1-2}
\medskip\noindent\textbf{4.1.2.}\quad
Let $Y$ be an object of $\operatorname{comp}(\mathcal A)$. Set

\VerdierSemanticAnchor{ch4d-4-1-2-1}
\begin{equation}
q\in\mathbb Z,\qquad
Y(q)=0\longrightarrow0\longrightarrow Y^{-q}\longrightarrow
Y^{-q+1}\longrightarrow Y^{-q+2}\longrightarrow\cdots,
\tag{4.1.2.1}
\end{equation}
where the differentials of this complex are induced by the differentials of
$Y$. Set likewise

\VerdierSemanticAnchor{ch4d-4-1-2-2}
\begin{equation}
Y(-\infty)=0,\qquad Y(+\infty)=Y.
\tag{4.1.2.2}
\end{equation}
There are monomorphisms

\VerdierSemanticAnchor{ch4d-4-1-2-filtration-chain}
\[
0=Y(-\infty)\hookrightarrow\cdots\hookrightarrow Y(q)
\hookrightarrow Y(q+1)\hookrightarrow Y(q+2)\hookrightarrow\cdots
\hookrightarrow Y(+\infty)=Y,
\]
