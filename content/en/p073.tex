\noindent\textbf{2.3.4.}\quad
Let $\varphi$ be the unique map $[n]\xrightarrow{\varphi}[1]$, and let
the group $\mathbb Z[n]$ act on $\operatorname{comp}(C)$ through
$\varphi\colon\mathbb Z[n]\longrightarrow\mathbb Z$. There is then, by
(1.5.2), a unique $\mathbb Z[n]$-functor twisted by the Koszul cocycle:

\VerdierSemanticAnchor{ch1-2-3-4-eq1}
\begin{equation}
  \left(\int_n^\Sigma,p_n\right)\colon
  \operatorname{comp}^{n}(C)\longrightarrow\operatorname{comp}(C)
  \tag{2.3.4.1}
\end{equation}

\VerdierSemanticAnchor{ch1-2-3-4-characterization}
such that:
\begin{equation*}
  p_n(e_i)=p_{i,\varphi}.
\end{equation*}

\VerdierSemanticAnchor{ch1-2-3-4-notation}
Hereafter, the notation $\int_n^\Sigma$ will denote the functor
$\int_n^\Sigma$ equipped with the twisted $\mathbb Z[n]$-functor structure
described above.

\VerdierSemanticAnchor{ch1-2-3-5}
\noindent\textbf{2.3.5.}\quad
The considerations developed in paragraphs (2.3.2), (2.3.3), and (2.3.4)
apply, with the obvious modifications, to the functors $\int_\varphi^\Pi$.

\VerdierSemanticAnchor{ch1-2-3-5-summability-restriction}
When countable sums and products are not necessarily representable in $C$,
the results above apply to the functors $\int_\varphi$ upon restricting to
complexes having suitable summability properties.

\VerdierSemanticAnchor{ch1-2-4}
\subsection{Extension of functors to complexes}

\VerdierSemanticAnchor{ch1-2-4-1}
\noindent\textbf{2.4.1.}\quad
There is a canonical functor:

\VerdierSemanticAnchor{ch1-2-4-1-eq1}
\begin{equation}
  \iota_n\colon
  \operatorname{comp}^{n}(C)^\circ
  \longrightarrow
  \operatorname{comp}^{n}(C^\circ)
  \tag{2.4.1.1}
\end{equation}
defined by the following formulas:

\VerdierSemanticAnchor{ch1-2-4-1-object-formula}
\begin{equation*}
  (\iota_nX)^\sigma=X^{-\sigma},
  \qquad X\in\operatorname{Ob}\bigl(\operatorname{comp}^{n}(C)\bigr),
\end{equation*}

\VerdierSemanticAnchor{ch1-2-4-1-eq2}
\begin{equation}
  d_{\iota_nX}^{\sigma,i}=d_X^{-\sigma-e_i,i},
  \tag{2.4.1.2}
\end{equation}

\VerdierSemanticAnchor{ch1-2-4-1-morphism-formula}
\begin{equation*}
  (\iota_nf)^\sigma=f^{-\sigma},
  \qquad f\in\operatorname{Fl}\bigl(\operatorname{comp}^{n}(C)\bigr).
\end{equation*}

\VerdierSemanticAnchor{ch1-2-4-1-graded-isomorphism}
The functor $\iota_n$ is an isomorphism of $\mathbb Z[n]$-categories
(the $\mathbb Z[n]$-category structure on
$\operatorname{comp}^{n}(C^\circ)$ is defined by (2.3.1); the graded
structure on $\operatorname{comp}^{n}(C)^\circ$ is defined by (2.3.1) and
(1.3.9)).

\VerdierSemanticAnchor{ch1-2-4-1-sigma-opposite-setup}
Suppose that countable direct sums are representable in $C$, and let
$\varphi\colon[n]\longrightarrow[m]$ be a map. Let us denote by:

\VerdierSemanticAnchor{ch1-2-4-1-eq3}
\begin{equation}
  \left(\int_{\varphi,C}^{\Sigma}\right)^\circ\colon
  \operatorname{comp}^{n}(C)^\circ
  \longrightarrow
  \operatorname{comp}^{m}(C)^\circ,
  \tag{2.4.1.3}
\end{equation}
