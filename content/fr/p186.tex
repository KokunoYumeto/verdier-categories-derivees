\VerdierSemanticAnchor{ch3d-2}
\section{Résolutions.}

\noindent\textit{Dans tout ce paragraphe, $\mathcal A$ désigne une catégorie
abélienne.}

\VerdierSemanticAnchor{ch3d-2-1}
\subsection{Morphismes élémentaires.}

\VerdierSemanticAnchor{ch3d-2-1-1}
\noindent\textbf{2.1.1.}\quad
Soient $X$ un complexe de $\mathcal A$ et $X^i$ le composant de degré $i$.
Soit $u\colon Y\longrightarrow X^i$ un morphisme de $\mathcal A$.
Construisons le diagramme commutatif ci-après :

\VerdierSemanticAnchor{ch3d-2-1-1-diagram}
\begin{equation}
\begin{tikzcd}[column sep=1.45em,row sep=2.25em]
{} & {} & X'^{i-1} \arrow[r,"v_2"] \arrow[d,"u'"']
  & Y \arrow[d,"u"] \arrow[dr,"v_3"] & {} & {} \\
X^{i-3} \arrow[r,"{d_X^{i-3}}"]
  & X^{i-2} \arrow[r,"{d_X^{i-2}}"] \arrow[ur,"v_1"]
  & X^{i-1} \arrow[r,"{d_X^{i-1}}"]
  & X^i \arrow[r,"{d_X^i}"]
  & X^{i+1} \arrow[r,"{d_X^{i+1}}"]
  & X^{i+2}
\end{tikzcd}
\tag{2.1.1.1}
\end{equation}
où le carré :

\VerdierSemanticAnchor{ch3d-2-1-1-cartesian-square}
\begin{center}
\begin{tikzcd}[column sep=4.2em,row sep=2.5em]
X'^{i-1} \arrow[r,"v_2"] \arrow[d,"u'"']
  & Y \arrow[d,"u"] \\
X^{i-1} \arrow[r,"{d_X^{i-1}}"'] & X^i
\end{tikzcd}
\end{center}
est cartésien et où les morphismes $v_1$ et $v_3$ sont définis par les
égalités :

\VerdierSemanticAnchor{ch3d-2-1-1-relations}
\[
v_2v_1=0,
\qquad
u'v_1=d_X^{i-2},
\qquad
v_3=d_X^iu.
\]
Désignons alors par $X(u,i)$ le complexe :

\VerdierSemanticAnchor{ch3d-2-1-1-complex}
\begin{equation}
\begin{aligned}
X(u,i)={}&\cdots\longrightarrow X^{i-3}
\xrightarrow{d_X^{i-3}}X^{i-2}
\xrightarrow{v_1}X'^{i-1}
\xrightarrow{v_2}Y\\
&\xrightarrow{v_3}X^{i+1}
\xrightarrow{d_X^{i+1}}X^{i+2}\longrightarrow\cdots,
\end{aligned}
\tag{2.1.1.2}
\end{equation}
et par $p(u,i)\colon X(u,i)\longrightarrow X$ le morphisme de
complexe :

\VerdierSemanticAnchor{ch3d-2-1-1-projection}
\begin{equation}
\begin{aligned}
p(u,i)^j&=\operatorname{identité},
  &&\text{pour }j\neq i-1,i,\\
p(u,i)^{i-1}&=u',\\
p(u,i)^i&=u.
\end{aligned}
\tag{2.1.1.3}
\end{equation}
\endgroup
