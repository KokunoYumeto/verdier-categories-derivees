By definition or by immediate verification, the property $P(q,q')$
is true in the following cases:

\begin{equation}
  \begin{gathered}
    q=0,\\
    q'=0,\\
    q\leq 0\ \text{and}\ q'=-1,\\
    q\geq 0\ \text{and}\ q'=1,\\
    q=1\ \text{and}\ q'=-1,\\
    q=-1\ \text{and}\ q'=1.
  \end{gathered}
  \tag{1.5.3.4}
\end{equation}

\VerdierSemanticAnchor{ch1-1-5-3-eq5}
Moreover, inspection of the different diagrams $D(q,q')$ that can be
formed from three integers $q,q',q''$ immediately shows (1.5.3.5):

\begin{quote}
  \noindent\textup{(1.5.3.5)}\quad
  \emph{If three of the four properties
  $P(q',q'')$, $P(q+q',q'')$, $P(q,q'+q'')$, $P(q,q')$ are true,
  then so is the fourth.}
\end{quote}

\VerdierSemanticAnchor{ch1-1-5-3-conclusion}
It then follows from (1.5.3.4) that the property $P(q,q')$ is true
for every pair of integers $q,q'$, which completes the proof in the
case $n=1$.

\VerdierSemanticAnchor{ch1-1-5-4}
\noindent\textbf{1.5.4. Existence in the general case.}\quad
Let $\mathbb Z^{n-1}\hookrightarrow\mathbb Z^n$ denote the subgroup
generated by the first $n-1$ elements of the basis of $\mathbb Z^n$;
then $\mathbb Z^n=\mathbb Z^{n-1}\oplus\mathbb Z e_n$.
Using the induction hypothesis, there exist isomorphisms

\VerdierSemanticAnchor{ch1-1-5-4-initial-isomorphisms}
\[
  \begin{aligned}
    p(\sigma)&\colon FT(\sigma)\longrightarrow T'(\sigma)F,
      &\sigma&\in\mathbb Z^{n-1},\\
    p(\tau)&\colon FT(\tau)\longrightarrow T'(\tau)F,
      &\tau&\in\mathbb Z e_n,
  \end{aligned}
\]
such that the diagrams

\VerdierSemanticAnchor{ch1-1-5-4-eq1}
\begin{equation}
  \resizebox{0.76\textwidth}{!}{%
    $\begin{tikzcd}[ampersand replacement=\&,column sep=7.0em,row sep=3.8em]
      FT(\lambda+\mu)
        \arrow[r,"{p(\lambda+\mu)}"]
        \arrow[d,"{F*c(\mu,\lambda)}"']
        \& T'(\lambda+\mu)F
        \arrow[dd,"{c'(\mu,\lambda)*F}"] \\
      FT(\lambda)T(\mu)
        \arrow[d,"{p(\lambda)*T(\mu)}"']
        \& {} \\
      T'(\lambda)FT(\mu)
        \arrow[r,"{T'(\lambda)*p(\mu)}"']
        \& T'(\lambda)T'(\mu)F
    \end{tikzcd}$%
  }
  \tag*{\(\textup{(1.5.4.1)}_{\lambda,\mu}\)}
\end{equation}
