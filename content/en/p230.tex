\noindent and then:

\VerdierSemanticAnchor{ch4d-4-1-6-6}
\begin{equation}
\int_n f\bigl((p,q),(p',q')\bigr)
=h\bigl((p,q),(p',q')\bigr).
\tag{4.1.6.6}
\end{equation}

Also let

\VerdierSemanticAnchor{ch4d-4-1-6-7}
\begin{equation}
{}^{m}\!d(p,q,r)\colon{}^{m}\!Y(q,r)
\longrightarrow{}^{m}\!Y(p,q)[1]
\tag{4.1.6.7}
\end{equation}

be the composite morphism

\VerdierSemanticAnchor{ch4d-4-1-6-8}
\begin{equation}
\begin{aligned}
{}^{m}\!Y(q,r)
&\xrightarrow{\int_n g(p,q,r)}
\int_n T_m\bigl({}^{m}\!X(p,q)\bigr)
\xrightarrow{(*)}
\left(\int_n{}^{m}\!X(p,q)\right)[1]\\
&=Y(p,q)[1].
\end{aligned}
\tag{4.1.6.8}
\end{equation}

where the morphism $(*)$ is the isomorphism expressing commutation of the
associated simple-complex functor with translations (Chapter~I, 2.3.2.1).

We thus obtain a filtration of the simple complex $Y=\int_n X$:

\VerdierSemanticAnchor{ch4d-4-1-6-9}
\begin{equation}
0={}^{m}\!Y(-\infty)\subset\cdots\subset{}^{m}\!Y(q)
\subset\cdots\subset{}^{m}\!Y(+\infty)=Y
\tag{4.1.6.9}
\end{equation}

called the \emph{filtration defined by the $m$-th degree of $X$}.

By the procedure recalled in (4.1.1), one may associate with the filtered
complex (4.1.6.9) a spectral object with values in $K(\mathcal A)$, denoted

\VerdierSemanticAnchor{ch4d-4-1-6-10}
\begin{equation}
\bigl(\mathcal Y(p,q),\delta(p,q,r)\bigr).
\tag{4.1.6.10}
\end{equation}

Finally, by an abuse of notation, in the proposition below we shall denote
by the same notation a morphism of $\operatorname{comp}(\mathcal A)$ and
its image in $K(\mathcal A)$.

\VerdierSemanticAnchor{ch4d-4-1-7}
\medskip\noindent\textbf{Proposition 4.1.7.}\quad

\VerdierSemanticAnchor{ch4d-4-1-7-a}
\noindent\textit{a) The objects ${}^{m}\!Y(p,q)$ (4.1.6.5), the morphisms
$h\bigl((p,q),(p',q')\bigr)$ (4.1.6.6), and the morphisms
${}^{m}\!d(p,q,r)$ (4.1.6.7) define a spectral object with values in
$K(\mathcal A)$ that depends functorially on the $n$-fold complex $X$.}

\noindent\textit{This spectral object is called the spectral object with
values in $K(\mathcal A)$ defined by the complex $X$ filtered by its
$m$-th degree.}

\VerdierSemanticAnchor{ch4d-4-1-7-b}
\noindent\textit{b) There is a canonical isomorphism (i.e.\ one that is
functorial with respect to the object $X$ of
$\operatorname{comp}^n(\mathcal A)$) between the spectral objects
$\bigl(\mathcal Y(p,q),\delta(p,q,r)\bigr)$ (4.1.6.10) and
$\bigl({}^{m}\!Y(p,q),{}^{m}\!d(p,q,r)\bigr)$.}
