\section{First Canonical Spectral Object.}

\VerdierSemanticAnchor{ch4d-4-1-1}
\noindent\textbf{4.1.1.}\quad
Recall that $\widetilde{\mathbb Z}$ denotes the ordered set
$\mathbb Z\cup\{-\infty,+\infty\}$, or equivalently the category associated
with this ordered set (Chapter II, 4.3.1). Let $\mathcal A$ be an additive
category and let $Y\colon\widetilde{\mathbb Z}\longrightarrow
\operatorname{comp}(\mathcal A)$ be a functor:

\VerdierSemanticAnchor{ch4d-4-1-1-1}
\begin{equation}
\begin{aligned}
q&\in\widetilde{\mathbb Z},
  &q&\longmapsto Y(q),\\
(p,q)&\in\mathcal{F}\ell_1(\widetilde{\mathbb Z}),
  &u(p,q)&\colon Y(p)\longrightarrow Y(q).
\end{aligned}
\tag{4.1.1.1}
\end{equation}

In Chapter I, (3.3.6), we thus associated with the functor $Y$ a spectral
object (of type $\widetilde{\mathbb Z}$) with values in $K(\mathcal A)$
(Chapter II, 4.1.2 and 4.1.3); we shall recall its construction.

First, for every element $(p,q)$ of
$\mathcal{F}\ell_1(\widetilde{\mathbb Z})$ (i.e.\ for every pair
$p,q\in\widetilde{\mathbb Z}$ such that $p\leq q$), let $\mathcal Y(p,q)$
be the cone of the morphism $u(p,q)$ (Chapter I, 3.1.2):

\VerdierSemanticAnchor{ch4d-4-1-1-2}
\begin{equation}
\begin{aligned}
\mathcal Y(p,q)^i&=Y(q)^i\oplus Y(p)^{i+1},\\
d_{\mathcal Y(p,q)}^i
&=
\begin{pmatrix}
d_{Y(q)}^i & u(p,q)^{i+1}\\
0 & -d_{Y(p)}^{i+1}
\end{pmatrix}.
\end{aligned}
\tag{4.1.1.2}
\end{equation}

For every pair $(p,q)$ and $(p',q')$ of elements of
$\mathcal{F}\ell_1(\widetilde{\mathbb Z})$ such that $p\leq p'$ and
$q\leq q'$, define

\VerdierSemanticAnchor{ch4d-4-1-1-3}
\begin{equation}
\Phi\bigl((p,q),(p',q')\bigr)\colon
\mathcal Y(p,q)\longrightarrow\mathcal Y(p',q'),
\tag{4.1.1.3}
\end{equation}
to be the homotopy class of the morphism of complexes

\VerdierSemanticAnchor{ch4d-4-1-1-4}
\begin{equation}
\Psi\bigl((p,q),(p',q')\bigr)\colon
\mathcal Y(p,q)\longrightarrow\mathcal Y(p',q'),
\tag{4.1.1.4}
\end{equation}
defined by the formulas

\VerdierSemanticAnchor{ch4d-4-1-1-5}
\begin{equation}
\Psi\bigl((p,q),(p',q')\bigr)^i
=
\begin{pmatrix}
u(q,q')^i & 0\\
0 & u(p,p')^{i+1}
\end{pmatrix}.
\tag{4.1.1.5}
\end{equation}
