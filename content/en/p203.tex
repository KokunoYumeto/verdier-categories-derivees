\VerdierSemanticAnchor{ch3d-2-4-2}
\medskip\noindent\textbf{Corollary 2.4.2.}\quad
\textit{Let $U$ be a universe. Suppose that $\mathcal A$ is a
$U$-category, i.e.\ that for every pair $Y$ and $Z$ of objects of $\mathcal A$,
$\operatorname{Hom}_{\mathcal A}(Y,Z)$ is an element of $U$, and that
direct sums indexed by elements of $U$ are representable in $\mathcal A$.
Let $(X_i)_{i\in I\in U}$ be a family of generators of $\mathcal A$, and
let $M$ be the set of objects of $\mathcal A$ of the form
$\coprod_{j\in J\in U}X_j$, where the $X_j$ are elements of the set
$\bigcup_{i\in I}\{X_i\}$. Then the natural functor}
\[
K^{-}(M)/\operatorname{Ac}^{-}(M)\longrightarrow D^{-}(\mathcal A)
\]
\noindent\textit{is an equivalence of categories.}

\VerdierSemanticAnchor{ch3d-2-4-2-proof}
This follows immediately from (2.4.1).

\VerdierSemanticAnchor{ch3d-2-4-3}
\medskip\noindent\textbf{Proposition 2.4.3.}\quad
\textit{Let $\mathcal A$ be an abelian category and $M$ a set of objects
of $\mathcal A$ having properties $(E_4)$ and $(E_5)$, such that every
object of $\mathcal A$ is a quotient of an element of $M$ and there is an
integer $n\in\mathbb N$ bounding the $M$-dimension of every object $X$ of
$\mathcal A$.}

\noindent\textit{a) The category $K^{*}(M)$ is left localizing for
$\operatorname{Ac}^{*}(\mathcal A)$ ($*=-,+,b,{}$ ``empty'').}

\noindent\textit{b) The natural functor}
\[
K^{*}(M)/\operatorname{Ac}^{*}(M)\longrightarrow D^{*}(\mathcal A)
\]
\noindent\textit{is an equivalence of categories.}

\VerdierSemanticAnchor{ch3d-2-4-3-proof}
This is a paraphrase of Proposition (2.3.3) ($*=+,b,{}$ ``empty''), or of
Proposition (2.3.1) ($*=-$).

\VerdierSemanticAnchor{ch3d-2-4-4}
\medskip\noindent\textbf{Proposition 2.4.4.}\quad
\textit{Denote by
$Q^{*}\colon K^{*}(\mathcal A)\longrightarrow D^{*}(\mathcal A)$ the
canonical functor.}

\noindent\textit{a) Let $X$ be an object of $\mathcal A$. The object $X$
is $Q$-free on the left (Chap. II, 2.3.2) if and only if $X$ is a
projective object of the category $\mathcal A$.}
