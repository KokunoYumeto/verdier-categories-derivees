\subsection{\texorpdfstring{$\mathbb Z^n$}{Z n}-Categories}

\VerdierSemanticAnchor{ch1-1-5-1}
\noindent\textbf{1.5.1.}\quad
Let $(\mathcal C,\mathbb Z^n,T,c)$ be a $\mathbb Z^n$-category,
$(\mathcal C',\mathbb Z^n,T',c')$ an additive
$\mathbb Z^n$-category, and
\[
  (F,p)\colon(\mathcal C,\mathbb Z^n,T,c)
  \longrightarrow(\mathcal C',\mathbb Z^n,T',c')
\]
a $\mathbb Z^n$-functor twisted by a normalized cocycle
$\alpha\in Z^2(\mathbb Z^n,\mathbb Z/2\mathbb Z)$. Let
$e_1,\ldots,e_n$ denote the canonical basis of $\mathbb Z^n$. Thus,
for every $i\in[1,n]$, we have an isomorphism of functors

\VerdierSemanticAnchor{ch1-1-5-1-eq1}
\begin{equation}
  p(e_i)\colon FT(e_i)\longrightarrow T'(e_i)F
  \tag{1.5.1.1}
\end{equation}
and, for every pair $(i,j)$ with $i\ne j$, an
$\alpha(e_i,e_j)\alpha^{-1}(e_j,e_i)$-commutative diagram:

\VerdierSemanticAnchor{ch1-1-5-1-eq2}
\begin{center}
  \resizebox{0.98\textwidth}{!}{%
    $\begin{tikzcd}[ampersand replacement=\&,column sep=4.2em,row sep=3.3em]
      FT(e_i+e_j)
        \arrow[r,"{F*c(e_i,e_j)}"]
        \arrow[d,"{F*c(e_j,e_i)}"]
        \&
      FT(e_j)T(e_i)
        \arrow[r,"{p(e_j)*T(e_i)}"]
        \&
      T'(e_j)FT(e_i)
        \arrow[d,"{T'(e_j)*p(e_i)}"'] \\
      FT(e_i)T(e_j)
        \arrow[d,"{p(e_i)*T(e_j)}"]
        \& {} \&
      T'(e_j)T'(e_i)F
        \arrow[d,"{c'(e_i,e_j)^{-1}*F}"'] \\
      T'(e_i)FT(e_j)
        \arrow[r,"{T'(e_i)*p(e_j)}"']
        \&
      T'(e_i)T'(e_j)F
        \arrow[r,"{c'(e_j,e_i)^{-1}*F}"']
        \&
      T'(e_i+e_j)F
    \end{tikzcd}$%
  }\;.
  \hfill\textup{(1.5.1.2)}
\end{center}

\VerdierSemanticAnchor{ch1-1-5-2}
\noindent\textbf{Proposition 1.5.2.}\quad
\emph{Let $F\colon\mathcal C\to\mathcal C'$ be a functor and, for
every $i\in[1,n]$, let
\[
  p_i\colon FT(e_i)\longrightarrow T'(e_i)F
\]
be an isomorphism of functors such that, for every pair $i,j$ with
$i\ne j$, the diagram analogous to diagram (1.5.1.2), with the
$p(e_i)$ replaced by the $p_i$, is
$\alpha(e_i,e_j)\alpha^{-1}(e_j,e_i)$-commutative. There exists one
and only one $\mathbb Z^n$-functor twisted by the cocycle $\alpha$:
\[
  (F,p)\colon(\mathcal C,\mathbb Z^n,T,c)
  \longrightarrow(\mathcal C',\mathbb Z^n,T',c')
\]
such that
\[
  p(e_i)=p_i,
  \qquad i\in[1,n].
\]}

\VerdierSemanticAnchor{ch1-1-5-2-proof-intro}
The proof of uniqueness is obtained by examining diagrams (1.4.3.11).
We leave the verification of this point to the reader. We shall limit
ourselves to giving some indications for the proof of existence. This
is carried out by induction on the integer $n$.
