\VerdierSemanticAnchor{ch1-1-4}
\subsection{Change of Groups}

\VerdierSemanticAnchor{ch1-1-4-1}
\noindent\textbf{1.4.1.}\quad
Let $f\colon H\to G$ be a group homomorphism. To every $G$-category
$(\mathcal C,G,T,c)$ we associate an $H$-category
$(\mathcal C,H,T_f,c_f)$ by letting $H$ act on $\mathcal C$ through
$f$, \emph{i.e.} by putting

\VerdierSemanticAnchor{ch1-1-4-1-eq1}
\begin{equation}
  \begin{aligned}
    T_f(h)&=T\bigl(f(h)\bigr),
      & h&\in H,\\
    c_f(h,h')&=c\bigl(f(h),f(h')\bigr),
      & h,h'&\in H.
  \end{aligned}
  \tag{1.4.1.1}
\end{equation}

\VerdierSemanticAnchor{ch1-1-4-1-gfunctor}
Similarly, given a $G$-functor
\[
  (F,m)\colon(\mathcal C,G,T,c)
  \longrightarrow(\mathcal C',G,T',c'),
\]
we associate to it an $H$-functor $(F,m_f)$ between the corresponding
$H$-categories by putting

\VerdierSemanticAnchor{ch1-1-4-1-eq2}
\begin{equation}
  m_f(h)=m\bigl(f(h)\bigr).
  \tag{1.4.1.2}
\end{equation}

\VerdierSemanticAnchor{ch1-1-4-1-morphisms}
Finally, a morphism of $G$-functors compatible with the operations of
$G$ is compatible with the operations of $H$.

\VerdierSemanticAnchor{ch1-1-4-1-eq3}
The preceding evidently defines a functor, which we call the
\emph{change-of-groups functor}:
\begin{equation}
  f^*\colon G\text{-}\operatorname{Cat}
  \longrightarrow H\text{-}\operatorname{Cat}.
  \tag{1.4.1.3}
\end{equation}

\VerdierSemanticAnchor{ch1-1-4-1-eq4}
More precisely, let $\mathcal C$ and $\mathcal C'$ be two
$G$-categories; the functor $f^*$ defines a map
\begin{equation}
  \begin{aligned}
    f^*(\mathcal C,\mathcal C')\colon\quad
      &\operatorname{Hom}_{G\text{-}\operatorname{Cat}}
        (\mathcal C,\mathcal C')\\
      &\longrightarrow
        \operatorname{Hom}_{H\text{-}\operatorname{Cat}}
        \bigl(f^*(\mathcal C),f^*(\mathcal C')\bigr).
  \end{aligned}
  \tag{1.4.1.4}
\end{equation}

\VerdierSemanticAnchor{ch1-1-4-1-homcategories}
But the sets
$\operatorname{Hom}_{G\text{-}\operatorname{Cat}}
(\mathcal C,\mathcal C')$ and
$\operatorname{Hom}_{H\text{-}\operatorname{Cat}}
\bigl(f^*(\mathcal C),f^*(\mathcal C')\bigr)$ are the sets of objects
of the categories
$\mathcal H\!om_{G\text{-}\operatorname{Cat}}
(\mathcal C,\mathcal C')$ and
$\mathcal H\!om_{H\text{-}\operatorname{Cat}}
\bigl(f^*(\mathcal C),f^*(\mathcal C')\bigr)$, whose morphisms are the
morphisms of functors compatible with the group operations, and the map
$f^*(\mathcal C,\mathcal C')$ extends naturally to a functor

\VerdierSemanticAnchor{ch1-1-4-1-eq5}
\begin{equation}
  \begin{aligned}
    f^*(\mathcal C,\mathcal C')\colon\quad
      &\mathcal H\!om_{G\text{-}\operatorname{Cat}}
        (\mathcal C,\mathcal C')\\
      &\longrightarrow
        \mathcal H\!om_{H\text{-}\operatorname{Cat}}
        \bigl(f^*(\mathcal C),f^*(\mathcal C')\bigr).
  \end{aligned}
  \tag{1.4.1.5}
\end{equation}

\VerdierSemanticAnchor{ch1-1-4-1-faithful}
This functor is faithful.
