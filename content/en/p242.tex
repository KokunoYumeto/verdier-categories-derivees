\VerdierSemanticAnchor{ch4d-4-5-4}
\medskip\noindent\textbf{Theorem 4.5.4.}\quad
\textit{Let
$F_1,F_2\colon D^*(\mathcal A)\longrightarrow\mathcal B$
($*=+,-,b,{}$ ``empty'') be two cohomological functors (respectively, let
$G_1,G_2\colon D^*(\mathcal A)\longrightarrow D^{*'}(\mathcal B)$ be two
exact functors), and let $m\colon F_1\longrightarrow F_2$ (respectively,
$m\colon G_1\longrightarrow G_2$) be a morphism of functors (respectively,
a morphism of exact functors (Chapter~I, 1.6.6.4)). The full subcategory of
$D^*(\mathcal A)$ defined by the objects $X$ of $D^*(\mathcal A)$ such that
$m(X[p])$ is an isomorphism for every integer $p$ is a strictly full and
saturated triangulated subcategory of $D^*(\mathcal A)$ (Chapter~II,
2.1.6), denoted $\operatorname{Is}(m)$.}

\VerdierSemanticAnchor{ch4d-4-5-4-a}
\noindent\textit{a) If every object of $\mathcal A$ is an object of
$\operatorname{Is}(m)$, then
$D^b(\mathcal A)\subset\operatorname{Is}(m)$.}

\VerdierSemanticAnchor{ch4d-4-5-4-b}
\noindent\textit{b) If every object of $\mathcal A$ is an object of
$\operatorname{Is}(m)$ and the functors $F_1$ and $F_2$ (respectively,
$G_1$ and $G_2$) are stationary on the right, then
$D^+(\mathcal A)\cap D^*(\mathcal A)\subset\operatorname{Is}(m)$.}

\VerdierSemanticAnchor{ch4d-4-5-4-c}
\noindent\textit{c) If every object of $\mathcal A$ is an object of
$\operatorname{Is}(m)$ and the functors $F_1$ and $F_2$ (respectively,
$G_1$ and $G_2$) are stationary on the left, then
$D^-(\mathcal A)\cap D^*(\mathcal A)\subset\operatorname{Is}(m)$.}

\VerdierSemanticAnchor{ch4d-4-5-4-d}
\noindent\textit{d) If every object of $\mathcal A$ is an object of
$\operatorname{Is}(m)$ and the functors $F_1$ and $F_2$ (respectively,
$G_1$ and $G_2$) are stationary, then
$D^*(\mathcal A)=\operatorname{Is}(m)$.}

\VerdierSemanticAnchor{ch4d-4-5-4-e}
\noindent\textit{e) If there is a subset $M$ of
$\operatorname{Ob}(\mathcal A)$ such that:}

\noindent\textit{$\alpha$) every object of $\mathcal A$ is isomorphic to a
quotient of an object belonging to $M$;}

\noindent\textit{$\beta$) every object belonging to $M$ is an object of
$\operatorname{Is}(m)$;}

\noindent\textit{and if, moreover, the functors $F_1$ and $F_2$
(respectively, $G_1$ and $G_2$) are stationary on the left, then
$D^-(\mathcal A)\cap D^*(\mathcal A)\subset\operatorname{Is}(m)$.}

\VerdierSemanticAnchor{ch4d-4-5-4-f}
\noindent\textit{f) If there is a subset $M$ of
$\operatorname{Ob}(\mathcal A)$ such that:}

\noindent\textit{$\alpha$) every object of $\mathcal A$ is isomorphic to a
subobject of an object belonging to $M$;}

\noindent\textit{$\beta$) every object belonging to $M$ is an object of
$\operatorname{Is}(m)$;}

\noindent\textit{and if, moreover, the functors $F_1$ and $F_2$
(respectively, $G_1$ and $G_2$) are stationary on the right, then
$D^+(\mathcal A)\cap D^*(\mathcal A)\subset\operatorname{Is}(m)$.}
