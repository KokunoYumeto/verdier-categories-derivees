\VerdierSemanticAnchor{ch3d-1-2-4}
\noindent\textbf{1.2.4.}\quad
The subcategory $K^{,b}(\mathcal A)/\operatorname{Ac}(\mathcal A)$
(respectively, $K^{,+}(\mathcal A)/\operatorname{Ac}(\mathcal A)$,
respectively, $K^{,-}(\mathcal A)/\operatorname{Ac}(\mathcal A)$) of
$D(\mathcal A)$ is denoted by $D^b(\mathcal A)$ (respectively,
$D^+(\mathcal A)$, respectively, $D^-(\mathcal A)$) and is called the
\emph{subcategory of bounded objects} (respectively, \emph{bounded below},
respectively, \emph{bounded above}). It should be noted that these
boundedness conditions apply only to the cohomology objects of the complexes
under consideration.

\VerdierSemanticAnchor{ch3d-1-2-5}
\medskip\noindent\textbf{1.2.5.}\quad
Let $X$ and $Y$ be two objects of $D(\mathcal A)$. We denote the group
$\operatorname{Hom}_{D(\mathcal A)}(X,Y)$ by
$\operatorname{Ext}_{\mathcal A}^{0}(X,Y)$. Likewise, we denote by
$\operatorname{Ext}_{\mathcal A}^{i}(X,Y)$ the groups
$\operatorname{Ext}_{\mathcal A}^{0}(X,Y[i])\simeq
\operatorname{Ext}_{\mathcal A}^{0}(X[-i],Y)$ (Chap. I, 1.3.8). This
notation will be justified later. Recall, however, that every distinguished
triangle $(X,X',X'',u,v,w)$ (respectively, $(Y,Y',Y'',u,v,w)$) in
$D(\mathcal A)$ gives an unbounded exact sequence:
\[
\cdots\longrightarrow \operatorname{Ext}_{\mathcal A}^{0}(X'',Y)
\longrightarrow \operatorname{Ext}_{\mathcal A}^{0}(X',Y)
\longrightarrow \operatorname{Ext}_{\mathcal A}^{0}(X,Y)
\longrightarrow \operatorname{Ext}_{\mathcal A}^{1}(X'',Y)
\longrightarrow\cdots
\]
(respectively,
\[
\cdots\longrightarrow \operatorname{Ext}_{\mathcal A}^{0}(X,Y)
\longrightarrow \operatorname{Ext}_{\mathcal A}^{0}(X,Y')
\longrightarrow \operatorname{Ext}_{\mathcal A}^{0}(X,Y'')
\longrightarrow \operatorname{Ext}_{\mathcal A}^{1}(X,Y)
\longrightarrow\cdots )
\]
(Chap. II, 1.2.1).

\VerdierSemanticAnchor{ch3d-1-2-6}
\medskip\noindent\textbf{1.2.6.}\quad
Let $M$ be a set of objects of $\mathcal A$. We denote by $D_M(\mathcal A)$
the full subcategory of $D(\mathcal A)$ defined by the complexes whose
cohomology objects belong to $M$ in every degree. We denote by
$D_M^*(\mathcal A)$ the full subcategory
$D_M(\mathcal A)\cap D^*(\mathcal A)$. If the full subcategory of
$\mathcal A$ defined by the objects of $M$ is a thick subcategory of
$\mathcal A$ (\textit{i.e.}, if it is abelian, if the inclusion functor is
exact, and if the set of objects of $M$ is stable under extensions), the
subcategories $D_M^*(\mathcal A)$ are full triangulated subcategories of
$D(\mathcal A)$.

\VerdierSemanticAnchor{ch3d-1-2-7}
\medskip\noindent\textbf{1.2.7.}\quad
Let $F\colon\mathcal A\longrightarrow\mathcal B$ be an exact functor between
two abelian categories and let
$K(F)\colon K(\mathcal A)\longrightarrow K(\mathcal B)$ be the extension of
$F$ to complexes up to homotopy (Chap. I, 2.5.10). The functor $K(F)$ sends
acyclic complexes of $\mathcal A$ to acyclic complexes of $\mathcal B$, is
exact (Chap. II, 1.3.7), and consequently (Chap. II, 2.2.6) defines an exact
functor $D(F)\colon D(\mathcal A)\longrightarrow D(\mathcal B)$. The functor
$D(F)$ sends bounded (respectively, bounded-below, respectively,
bounded-above) objects of $D(\mathcal A)$ to objects of the same kind in
$D(\mathcal B)$. Most often we shall use the abusive notation
$F\colon D(\mathcal A)\longrightarrow D(\mathcal B)$ for the functor
$D(F)$.

\VerdierSemanticAnchor{ch3d-1-2-8}
\medskip\noindent\textbf{1.2.8.}\quad
The canonical isomorphism (Chap. II, 1.3.8)
$i\colon K(\mathcal A)^o\longrightarrow K(\mathcal A^o)$ sends the objects
of $\operatorname{Ac}(\mathcal A)^o$ to objects of
$\operatorname{Ac}(\mathcal A^o)$ and therefore defines an isomorphism of
triangulated categories, again denoted by
$i\colon D(\mathcal A)^o\longrightarrow D(\mathcal A^o)$.
