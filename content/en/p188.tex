\VerdierSemanticAnchor{ch3d-2-1-4}
\medskip\noindent\textbf{Lemma 2.1.4.}\quad
\textit{Let $m\colon X\longrightarrow Y$ be a quasi-isomorphism in
$\operatorname{comp}(\mathcal A)$. There exists a sequence of morphisms}
\[
u_n\colon X_n\longrightarrow X_{n-1},\qquad n\in\mathbb Z,
\]
\noindent\textit{such that, for every integer $n$, $u_n$ is an elementary
morphism of height $-n$, together with morphisms}

\VerdierSemanticAnchor{ch3d-2-1-4-limits}
\[
\varinjlim_n X_n\xleftarrow{\sim}Y,
\qquad
\varprojlim_n X_n\xleftarrow{v}X,
\qquad v\text{ a homotopy equivalence (Chap. I, 2.5.11)},
\]
\noindent\textit{such that the diagram}

\VerdierSemanticAnchor{ch3d-2-1-4-square}
\begin{center}
\begin{tikzcd}[column sep=5.6em,row sep=3em]
X \arrow[r,"v"] \arrow[d,"m"']
  & \displaystyle\varprojlim_n X_n \arrow[d] \\
Y \arrow[r,"{\sim}"'] & \displaystyle\varinjlim_n X_n
\end{tikzcd}
\end{center}
\noindent\textit{is commutative up to homotopy.}

\VerdierSemanticAnchor{ch3d-2-1-4-proof}
First observe that for every morphism $m\colon X\longrightarrow Y$ in
$\operatorname{comp}(\mathcal A)$, there is an epimorphism
$m'\colon X'\longrightarrow Y$ and a diagram commutative up to homotopy:

\VerdierSemanticAnchor{ch3d-2-1-4-epimorphism-diagram}
\begin{center}
\begin{tikzcd}[column sep=3.5em,row sep=3.2em]
X \arrow[rr,"v"] \arrow[dr,"m"'] && X' \arrow[dl,"{m'}"] \\
& Y &
\end{tikzcd}\qquad,
\end{center}
