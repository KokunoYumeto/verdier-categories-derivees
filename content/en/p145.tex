\VerdierSemanticAnchor{ch2-2-3-2-right-left-terminology}
We shall say that an object $Y$ of $C$ is \emph{$F$-free on the right} if
there is an object $X$ of $C'$ such that $Y$ is free on the right over
$X$. We shall say that an object $X$ of $C'$ is \emph{$F$-liberable on the
right} if there is an object $Y$ of $C$ free on the right over $X$. By
passing to opposite categories, one defines analogously the notions of an
object free or liberable on the \emph{left}.

\noindent\textbf{Proposition 2.3.3.}\quad
Let $\mathcal D$ be a triangulated category, let $\mathcal B$ be a full
triangulated subcategory, let
$Q\colon\mathcal D\longrightarrow\mathcal D/\mathcal B$ be the quotient
functor (2.2.10), and let $S_{\mathcal D}(\mathcal B)$ be the
multiplicative system of $\mathcal D$ associated with $\mathcal B$ (2.1.8).

\VerdierSemanticAnchor{ch2-2-3-3-a}
\par\noindent a) For every object $Y$ of $\mathcal D$, the following
conditions are equivalent:

\VerdierSemanticAnchor{ch2-2-3-3-a-i}
\par\noindent\hspace{1.2em}i) The object $Y$ is $Q$-free on the right.

\VerdierSemanticAnchor{ch2-2-3-3-a-ii}
\par\noindent\hspace{1.2em}ii) For every morphism
$s\in S_{\mathcal D}(\mathcal B)$, $s\colon S\longrightarrow T$, the map
\[
  \operatorname{Hom}_{\mathcal D}(s,Y)\colon
  \operatorname{Hom}_{\mathcal D}(T,Y)\longrightarrow
  \operatorname{Hom}_{\mathcal D}(S,Y)
\]
is a bijection.

\VerdierSemanticAnchor{ch2-2-3-3-a-iii}
\par\noindent\hspace{1.2em}iii) For every object $B$ of $\mathcal B$,
$\operatorname{Hom}_{\mathcal D}(B,Y)=0$.

\VerdierSemanticAnchor{ch2-2-3-3-a-iv}
\par\noindent\hspace{1.2em}iv) Every morphism
$Y\xrightarrow{s}Z$, with $s\in S_{\mathcal D}(\mathcal B)$, admits a
retraction.

\VerdierSemanticAnchor{ch2-2-3-3-a-v}
\par\noindent\hspace{1.2em}v) For every object $X$ of $\mathcal D$, the map
\[
  \operatorname{Hom}_{\mathcal D}(X,Y)\longrightarrow
  \operatorname{Hom}_{\mathcal D/\mathcal B}(Q(X),Q(Y))
\]
is bijective.

\VerdierSemanticAnchor{ch2-2-3-3-b}
\par\noindent b) Let $L_R(Q)$ be the full subcategory of $\mathcal D$
defined by the objects that are $Q$-free on the right. The category
$L_R(Q)$ is triangulated, strictly full, and saturated.

\VerdierSemanticAnchor{ch2-2-3-3-c}
\par\noindent c) The restriction of the quotient functor to $L_R(Q)$,
\[
  Q|_{L_R(Q)}\colon L_R(Q)\longrightarrow\mathcal D/\mathcal B,
\]
is fully faithful and injective on objects.

\VerdierSemanticAnchor{ch2-2-3-3-d}
\par\noindent d) Let $\operatorname{Lib}_R(Q)\subset\mathcal D/\mathcal B$
be the full subcategory defined by the objects of $\mathcal D/\mathcal B$
that are $Q$-liberable on the right. The subcategory
$\operatorname{Lib}_R(Q)$ is triangulated and strictly full. If countable
products or countable direct sums are representable in $\mathcal D$, the
category $\operatorname{Lib}_R(Q)$ is saturated.

\VerdierSemanticAnchor{ch2-2-3-3-e}
\par\noindent e) The functor $Q$ sends $L_R(Q)$ into
$\operatorname{Lib}_R(Q)$. The category $\operatorname{Lib}_R(Q)$ is the
strictly full closure of the image of $L_R(Q)$. The freedom arrows are
isomorphisms.
