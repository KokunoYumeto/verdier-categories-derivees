\VerdierSemanticAnchor{ch3d-2-2-2}
\noindent\textbf{Lemma 2.2.2.}\quad
\textit{Let
$0\longrightarrow X\longrightarrow Y\longrightarrow Z\longrightarrow0$
be an exact sequence in $\mathcal A$.}

\noindent\textit{a) If $X$ and $Y$ are elements of $M^{\wedge}$, there is
an epimorphism $u\colon L\longrightarrow Z$, with $L\in M$.}

\noindent\textit{b) If $X$ and $Z$ are elements of $M^{\wedge}$, there is
an epimorphism $u\colon L\longrightarrow Y$, with $L\in M$.}

\noindent\textit{c) If $Y$ and $Z$ are elements of $M^{\wedge}$, there is
an epimorphism $u\colon L\longrightarrow X$, with $L\in M$.}

\VerdierSemanticAnchor{ch3d-2-2-3}
\medskip\noindent\textbf{Lemma 2.2.3.}\quad
\textit{Let
$0\longrightarrow X\longrightarrow Y\longrightarrow Z\longrightarrow0$
be an exact sequence in $\mathcal A$.}

\noindent\textit{a) If $Y$ and $Z$ are elements of $M^{\wedge}$, then for
every epimorphism $u\colon L\longrightarrow X$, with $L\in M$, the kernel
of $u$ is the kernel of an epimorphism between two elements of
$M^{\wedge}$.}

\noindent\textit{b) If $X$ and $Y$ are elements of $M^{\wedge}$, then for
every epimorphism $u\colon L\longrightarrow Z$, with $L\in M$, the kernel
of $u$ is the cokernel of a monomorphism between two elements of
$M^{\wedge}$.}

\noindent\textit{c) If $X$ and $Z$ are elements of $M^{\wedge}$, then for
every epimorphism $u\colon L\longrightarrow Y$, with $L\in M$, the kernel
of $u$ is the kernel of an epimorphism between two elements of
$M^{\wedge}$.}

\VerdierSemanticAnchor{ch3d-2-2-2-3-proof}
The proof of the two lemmas is left to the reader. Note that properties
(a) and (c) of Lemmas (2.2.2) and (2.2.3) do not use property $(E_2)$.

\VerdierSemanticAnchor{ch3d-2-2-4}
\medskip\noindent\textbf{Proposition 2.2.4.}\quad
\textit{a) The set $M^{\wedge}$ has properties $(E_6)$ and $(E_2)$. The
set $M$ is contained in $M^{\wedge}$. Every object of $M^{\wedge}$ is a
quotient of an object of $M$.}

\noindent\textit{b) Every set of objects $M'$ containing $M$, such that
every object of $M'$ is a quotient of an object of $M$ and which has
property $(E_3)$, is contained in $M^{\wedge}$.}

\noindent\textit{c) In particular,
$(M^{\wedge})^{\wedge}=M^{\wedge}$.}

\VerdierSemanticAnchor{ch3d-2-2-4-proof}
Property a) follows from the two preceding lemmas. Properties b) and c)
are immediate.

\VerdierSemanticAnchor{ch3d-2-2-5}
\medskip\noindent\textbf{2.2.5.}\quad
Let $M\subset M'$ be two sets of objects of $\mathcal A$ such that $M$
has properties $(E_4)$ and $(E_5)$, $M'$ has property $(E_4)$, and every
object of $M'$ is a quotient of an object of $M$. An object $X$ of $M'$
is said to have $M$-dimension less than or equal to $n$, written
$M\text{-}\dim(X)\leq n$, if there is an exact sequence

\VerdierSemanticAnchor{ch3d-2-2-5-exact-sequence}
\[
0\longrightarrow L_n\longrightarrow L_{n-1}\longrightarrow\cdots
\longrightarrow L_0\longrightarrow X\longrightarrow0,
\qquad L_i\in M,\quad0\leq i\leq n.
\]
