\VerdierSemanticAnchor{ch3-3-2-3-hom-map}
Let $(X,Y,f)$ and $(X',Y',f')$ be two objects of $\mathrm A(\mathcal D)$.
To prove Proposition (3.2.3), it suffices to show that the homomorphism
defined by the functor $\operatorname{Im}$
\[
\operatorname{Hom}_{\mathrm A(\mathcal D)}((X,Y,f),(X',Y',f'))
\longrightarrow
\operatorname{Hom}_{\mathcal D^\wedge}(\operatorname{Im}(f),\operatorname{Im}(f'))
\tag{3.2.3.2}
\]
is an isomorphism.

\VerdierSemanticAnchor{ch3-3-2-3-injectivity}
First prove injectivity. Let
$(g;g')\colon(X,Y,f)\to(X',Y',f')$ be a morphism of
$\mathcal{F}\ell_1(\mathcal D)$ such that
$\operatorname{Im}((g;g'))=0$. The composite morphism:
\[
\operatorname{Im}((f';\operatorname{id}_{Y'}))\circ
\operatorname{Im}((g;g'))\circ
\operatorname{Im}((\operatorname{id}_X;f))=h(g'f)=0.
\]
Since $h$ is fully faithful, $g'f=0$; hence $(g;g')$ is negligible.
Surjectivity of the homomorphism (3.2.3.2) follows from the next lemma.

\VerdierSemanticAnchor{ch3-3-2-4}
\medskip\noindent\textbf{Lemma 3.2.4.}\quad
\textit{Let
$\alpha\colon\operatorname{Im}(f)\to\operatorname{Im}(f')$ be a morphism of
$\mathcal D^\wedge$. There exist two morphisms of $\mathcal D$,
$g\colon X\to X'$ and $g'\colon Y\to Y'$, such that the following diagrams
commute:}

\VerdierSemanticAnchor{ch3-3-2-4-1}
\[
\begin{tikzcd}[column sep=6em,row sep=2.3em]
h(X)\arrow[r,"\operatorname{Im}((\operatorname{id}_X;f))"]\arrow[d,"h(g)"']&
\operatorname{Im}(f)\arrow[d,"\alpha"]\\
h(X')\arrow[r,"\operatorname{Im}((\operatorname{id}_{X'};f'))"']&
\operatorname{Im}(f')
\end{tikzcd}
\tag{3.2.4.1}
\]

\VerdierSemanticAnchor{ch3-3-2-4-2}
\[
\begin{tikzcd}[column sep=5em,row sep=2.3em]
h(X)\arrow[r,"h(f)"]\arrow[d,"h(g)"']&h(Y)\arrow[d,"h(g')"]\\
h(X')\arrow[r,"h(f')"']&h(Y')
\end{tikzcd}
\tag{3.2.4.2}
\]

\VerdierSemanticAnchor{ch3-3-2-4-consequence}
Indeed, the lemma gives $\alpha=\operatorname{Im}((g;g'))$ by (3.2.3.1)
and the fact that $\operatorname{Im}((\operatorname{id}_X;f))$ is an
epimorphism in $\mathcal D^\wedge$.
