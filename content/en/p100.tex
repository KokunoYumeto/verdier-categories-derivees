\VerdierSemanticAnchor{ch1-3-3-6}
\noindent\textbf{3.3.6.} Let
$F\colon \mathcal{J}\longrightarrow\operatorname{comp}(C)$ be a functor. Denote
by $\mathcal{F}\ell_n(\mathcal{J})$ the category of strings of $n$ composable
morphisms of $\mathcal{J}$: the objects of $\mathcal{F}\ell_n(\mathcal{J})$ are
the diagrams in $\mathcal{J}$

\VerdierSemanticAnchor{ch1-3-3-6-object-diagram}
\[
  i_1\xrightarrow{f_1}i_2\xrightarrow{f_2}i_3
  \longrightarrow\cdots\xrightarrow{f_n}i_{n+1}.
\]
\VerdierSemanticAnchor{ch1-3-3-6-morphism}

$\mathcal{J}$. A morphism in $\mathcal{F}\ell_n(\mathcal{J})$ from

\VerdierSemanticAnchor{ch1-3-3-6-cone-functor}
Let $i\xrightarrow{f}j$ be an object of $\mathcal{F}\ell_1(\mathcal{J})$.
Applying the functor $F$ gives a morphism in $\operatorname{comp}(C)$,
\[
  F(i)\xrightarrow{F(f)}F(j),
\]
which can be regarded as an object of $\operatorname{Tr}_1(C)$; hence applying
the cone functor gives a functor
$S\colon\mathcal{F}\ell_1(\mathcal{J})\longrightarrow\mathrm{K}(C)$:

\VerdierSemanticAnchor{ch1-3-3-6-1}
\begin{equation}
  S(f)=S\bigl(i\xrightarrow{f}j\bigr)
  =c\bigl(F(i)\xrightarrow{F(f)}F(j)\bigr).
  \tag{3.3.6.1}
\end{equation}
