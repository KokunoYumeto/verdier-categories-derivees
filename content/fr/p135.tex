\VerdierSemanticAnchor{ch2-2-2-1-localized-hom-definition}
\smallskip
\noindent Pour tout couple $X,Y$ d'objets de $\mathcal C$, posons :

\VerdierSemanticAnchor{ch2-2-2-1-formula-2-2-1-1}
\par\smallskip
\noindent\makebox[0pt][l]{(2.2.1.1)}%
\begin{minipage}[t]{\linewidth}
  \centering
  $\displaystyle
    \operatorname{Hom}_{\mathcal C(S^{-1})}(X,Y)
    =\varinjlim_{(S/X)^{\circ}}
     \operatorname{Hom}_{\mathcal C}(\,\cdot\,,Y)\,.$
\end{minipage}
\par\smallskip

\VerdierSemanticAnchor{ch2-2-2-1-dot-explanation}
\noindent (Le “point” dans la formule (2.2.1.1) désigne la source d'un
objet variable de $S/X$).

\VerdierSemanticAnchor{ch2-2-2-1-filtered-category}
Il résulte immédiatement des propriétés SM1), SM2) et SM3) que la catégorie
$(S/X)^{\circ}$ est filtrante [6].
\VerdierSemanticAnchor{ch2-2-2-1-roof-class}%
Il en résulte qu'un élément de
$\operatorname{Hom}_{\mathcal C(S^{-1})}(X,Y)$ est une classe de diagrammes
du type :

\VerdierSemanticAnchor{ch2-2-2-1-roof-formula}
\[
  D=X\xleftarrow{s}X'\xrightarrow{m}Y,\qquad s\in S,
\]

\VerdierSemanticAnchor{ch2-2-2-1-equivalence-setup}
\noindent deux diagrammes
$D_1=X\xleftarrow{s_1}X_1\xrightarrow{m_1}Y$ et
$D_2=X\xleftarrow{s_2}X_2\xrightarrow{m_2}Y$ appartenant à la même classe
si et seulement s'il existe un diagramme :

\VerdierSemanticAnchor{ch2-2-2-1-common-refinement-roof}
\[
  D_3=X\xleftarrow{s_3}X_3\xrightarrow{m_3}Y,\qquad s_3\in S,
\]

\VerdierSemanticAnchor{ch2-2-2-1-common-refinement-diagram-introduction}
\noindent et un diagramme commutatif :

\VerdierSemanticAnchor{ch2-2-2-1-common-refinement-diagram}

\VerdierSemanticAnchor{ch2-2-2-1-q-of-f}
\noindent Pour tout élément $f\in\operatorname{Hom}_{\mathcal C}(X,Y)$,
on désigne par $Q(f)$ la classe du diagramme
$X\xleftarrow{\operatorname{id}_X}X\xrightarrow{f}Y$.
\VerdierSemanticAnchor{ch2-2-2-1-q-of-s-inverse}%
Pour tout morphisme $s\colon Y\to X$, $s\in S$, on désigne par $Q(s)^{-1}$
la classe du diagramme
$X\xleftarrow{s}Y\xrightarrow{\operatorname{id}_Y}Y$.

\[
  \prod_{(X,Y)\in\operatorname{Ob}(\mathcal C)\times\operatorname{Ob}(\mathcal C)}
  \operatorname{Hom}_{\mathcal C(S^{-1})}(X,Y),
