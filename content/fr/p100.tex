\VerdierSemanticAnchor{ch1-3-3-6}
\noindent\textbf{3.3.6.} Soit $F\colon \mathcal{J}\longrightarrow\operatorname{comp}(C)$ un
foncteur. Désignons par $\mathcal{F}\ell_n(\mathcal{J})$ la catégorie des suites de
$n$ morphismes composables de $\mathcal{J}$ : les objets de
$\mathcal{F}\ell_n(\mathcal{J})$ sont les diagrammes de $\mathcal{J}$ :

\VerdierSemanticAnchor{ch1-3-3-6-object-diagram}
\[
  i_1\xrightarrow{f_1}i_2\xrightarrow{f_2}i_3
  \longrightarrow\cdots\xrightarrow{f_n}i_{n+1}.
\]
\VerdierSemanticAnchor{ch1-3-3-6-morphism}
Soient $(i_p,f_p)$ et $(j_p,g_p)$ deux suites de $n$ morphismes composables de
$\mathcal{J}$. Un morphisme, dans $\mathcal{F}\ell_n(\mathcal{J})$, de
$(i_p,f_p)$ dans $(j_p,g_p)$ est un diagramme commutatif :

\VerdierSemanticAnchor{ch1-3-3-6-morphism-diagram}

\VerdierSemanticAnchor{ch1-3-3-6-composition}
La composition des morphismes se définit de la manière évidente. On désigne
par $\mathrm{Fl}_n(\mathcal{J})$ l'ensemble des objets de
$\mathcal{F}\ell_n(\mathcal{J})$.

\VerdierSemanticAnchor{ch1-3-3-6-cone-functor}
Soit $i\xrightarrow{f}j$ un objet de $\mathcal{F}\ell_1(\mathcal{J})$. En
appliquant le foncteur $F$, on obtient un morphisme de
$\operatorname{comp}(C)$ :
\[
  F(i)\xrightarrow{F(f)}F(j),
\]
qu'on peut interpréter comme un objet de $\operatorname{Tr}_1(C)$, d'où, en
appliquant le foncteur cône, un foncteur
$S\colon\mathcal{F}\ell_1(\mathcal{J})\longrightarrow\mathrm{K}(C)$ :

\VerdierSemanticAnchor{ch1-3-3-6-1}
\begin{equation}
  S(f)=S\bigl(i\xrightarrow{f}j\bigr)
  =c\bigl(F(i)\xrightarrow{F(f)}F(j)\bigr).
  \tag{3.3.6.1}
\end{equation}
