\VerdierSemanticAnchor{ch3-3-2}
\subsection{Properties of
  \texorpdfstring{$\mathrm A(\mathcal D)$}{A(D)}
  (\texorpdfstring{$\mathcal D$}{D} a triangulated category).}

\VerdierSemanticAnchor{ch3-3-2-0}
\noindent\textbf{3.2.0.}\quad
Let $\mathcal D$ and $\mathcal D'$ be two triangulated categories. Denote by
$\mathcal E x(\mathcal D,\mathcal D')$ the category whose objects are the
exact functors from $\mathcal D$ to $\mathcal D'$ and whose morphisms are
morphisms of exact functors. Likewise, denote by
$\mathbb Z\text{-}\operatorname{Add}(\mathrm A(\mathcal D),
\mathrm A(\mathcal D'))$ the category whose objects are the additive
$\mathbb Z$-functors from $\mathrm A(\mathcal D)$ to
$\mathrm A(\mathcal D')$ and whose morphisms are morphisms of
$\mathbb Z$-functors. It follows from the preceding discussion that we have
defined a functor

\VerdierSemanticAnchor{ch3-3-2-0-1}
\[
\mathrm A_{\mathcal D,\mathcal D'}\colon
\mathcal E x(\mathcal D,\mathcal D')
\longrightarrow
\mathbb Z\text{-}\operatorname{Add}
(\mathrm A(\mathcal D),\mathrm A(\mathcal D')),
\tag{3.2.0.1}
\]
which associates to an exact functor $F$ the functor $\mathrm A(F)$.

\VerdierSemanticAnchor{ch3-3-2-1}
\medskip
\noindent\textbf{Theorem 3.2.1.}\begingroup\renewcommand{\thefootnote}{(*)}%
\footnote{This theorem was communicated to me by Freyd [17].}\endgroup\quad
\textit{a) If $\mathcal D$ is a triangulated category, then
$\mathrm A(\mathcal D)$ is abelian. If
$F\colon\mathcal D\to\mathcal D'$ is an exact functor between triangulated
categories, then the functor $\mathrm A(F)$ (3.1.4.1) is exact. The functor
$\mathrm A_{\mathcal D,\mathcal D'}$ (3.2.0.1) is fully faithful and
injective on objects.}

\VerdierSemanticAnchor{ch3-3-2-1-b}
\textit{b) The objects of $\mathcal D$ are injective and projective in
$\mathrm A(\mathcal D)$. Every object of $\mathrm A(\mathcal D)$ is
isomorphic to a quotient object (respectively, a subobject) of an object of
$\mathcal D$.}

\VerdierSemanticAnchor{ch3-3-2-1-c}
\textit{c) Let $\mathcal A$ be an abelian category. Denote by
$\mathcal E x(\mathrm A(\mathcal D),\mathcal A)$ the category whose objects
are exact functors from $\mathrm A(\mathcal D)$ to $\mathcal A$ and whose
morphisms are morphisms of functors. Likewise, denote by
$\mathcal H\!om(\mathcal D,\mathcal A)$ the category whose objects are
functors from $\mathcal D$ to $\mathcal A$ and whose morphisms are morphisms
of functors. Composition with
$\mathcal H\colon\mathcal D\to\mathrm A(\mathcal D)$ defines a functor}
\[
\mathcal E x(\mathrm A(\mathcal D),\mathcal A)
\longrightarrow
\mathcal H\!om(\mathcal D,\mathcal A).
\]
\textit{This functor is fully faithful. Its essential image consists of the
cohomological functors from $\mathcal D$ to $\mathcal A$ (1.1.5). In
particular, $\mathcal H\colon\mathcal D\to\mathrm A(\mathcal D)$ is a
cohomological functor.}

\VerdierSemanticAnchor{ch3-3-2-1-proof-plan}
The proof of this theorem will occupy us through paragraph (3.2.9). We first prove
three intermediate propositions.

\VerdierSemanticAnchor{ch3-3-2-2}
\medskip
\noindent\textbf{3.2.2.}\quad
Let $U$ be a universe such that $\mathcal D$ is $U$-small, that is, such that
$\mathcal D$ is an element of $U$. Let
$\mathcal D^\wedge$ be the category of additive contravariant functors on
$\mathcal D$ with values in the category of $U$-small abelian groups. The category
$\mathcal D^\wedge$ is abelian, and the canonical functor
$h\colon\mathcal D\to\mathcal D^\wedge$, which associates to each object of
$\mathcal D$ the
functor it represents, is fully faithful. The functor $h$ sends every object
of $\mathcal D$ to a projective object of $\mathcal D^\wedge$.
