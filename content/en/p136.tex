\VerdierSemanticAnchor{ch2-2-2-2}
\smallskip
\noindent\textbf{Theorem 2.2.2.}\quad
\begingroup\itshape
\VerdierSemanticAnchor{ch2-2-2-2-a}%
a) There exists one and only one partially defined composition law on
$\operatorname{Fl}(\mathcal C(S^{-1}))$ such that:

\VerdierSemanticAnchor{ch2-2-2-2-a-1}
\par\noindent\hspace*{2em}1) $\mathcal C(S^{-1})$ is a category.

\VerdierSemanticAnchor{ch2-2-2-2-a-2}
\par\noindent\hspace*{2em}2) $Q(s)$ is an isomorphism and $Q(s)^{-1}$ is its
inverse $(s\in S)$.

\VerdierSemanticAnchor{ch2-2-2-2-a-3}
\par\noindent\hspace*{2em}3) For every diagram
$D=X\xleftarrow{s}X'\xrightarrow{m}Y$, $s\in S$, one has
$\overline D=Q(m)Q(s)^{-1}$
$(\overline D\in\operatorname{Hom}_{\mathcal C(S^{-1})}(X,Y)$ is the class
of $D$).

\VerdierSemanticAnchor{ch2-2-2-2-a-4}
\par\noindent\hspace*{2em}4) $Q$ is a functor.

\VerdierSemanticAnchor{ch2-2-2-2-b}
\par\noindent b) Every morphism of $\mathcal C(S^{-1})$ can be written as a
composite $Q(t)^{-1}Q(m)$, $m\in\operatorname{Fl}(\mathcal C)$, $t\in S$,
(resp. $Q(p)Q(s)^{-1}$, $p\in\operatorname{Fl}(\mathcal C)$, $s\in S$).

\VerdierSemanticAnchor{ch2-2-2-2-c}
\par\noindent c) The functor $Q$ commutes with finite projective limits and
finite inductive limits.

\VerdierSemanticAnchor{ch2-2-2-2-d}
\par\noindent d) For every category $\mathcal A$, composition with the
functor $Q$ defines a functor:

\VerdierSemanticAnchor{ch2-2-2-2-hom-q-a}
\[
  \mathcal{H}\!om(Q,\mathcal A)\colon
  \mathcal{H}\!om(\mathcal C(S^{-1}),\mathcal A)
  \longrightarrow \mathcal{H}\!om(\mathcal C,\mathcal A)
\]

\noindent\textup{($\mathcal{H}\!om$ denotes the category of functors).}
\VerdierSemanticAnchor{ch2-2-2-2-universal-image}%
The functor $\mathcal{H}\!om(Q,\mathcal A)$ is injective on objects, fully
faithful, and the image of
$\mathcal{H}\!om(\mathcal C(S^{-1}),\mathcal A)$ is the strictly full
subcategory of $\mathcal{H}\!om(\mathcal C,\mathcal A)$ defined by the functors from
$\mathcal C$ to $\mathcal A$ that carry the morphisms of $S$ to
isomorphisms.
\par\endgroup

\VerdierSemanticAnchor{ch2-2-2-2-proof-omitted}
The proof of this theorem is left to the reader, who may consult [16].
